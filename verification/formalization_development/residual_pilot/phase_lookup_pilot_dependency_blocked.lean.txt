/-- A phase multiplication is a small linear expression, before any huge integer substitution. -/
theorem phase2_coordinate0 (n : Fin 24 → Int) :
    Scalar.mulNumerator n (phaseNumerator 2) 0 = 2*n 0 + 10*n 1 - 40*n 19 := by
  norm_num [Scalar.mulNumerator, Scalar.mulTerms, phaseNumerator]
  ring

end CGLMP5.SOSFinite.PhasePilot
